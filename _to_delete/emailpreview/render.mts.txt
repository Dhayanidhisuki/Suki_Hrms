import fs from "node:fs";
import path from "node:path";
import { digestTemplate, testEmailTemplate } from "../../src/lib/email/emailTemplates.ts";
type DueCalibrationTool = any; type NotificationUser = any;

const d = (s: string) => new Date(s + "T00:00:00");
const today = d("2026-08-17");
const mk = (no: string, name: string, unit: any, loc: string | null, due: string): DueCalibrationTool => {
  const days = Math.round((d(due).getTime() - today.getTime()) / 86400000);
  return {
    refNo: 1, toolOrGaugeNo: no, name, description: null, size: "0-25mm",
    usedLocation: loc, unitCode: unit, responsibility: "Internal",
    dueDate: d(due), daysRemaining: days,
    dueStatus: days < 0 ? "OVERDUE" : days === 0 ? "DUE_TODAY" : "DUE_SOON",
  };
};

const items: DueCalibrationTool[] = [
  mk("MP-TRG-517 G6NG", "TRG", "UNIT1", "Production", "2026-08-12"),
  mk("MP-TRG-638", "TPG", "UNIT1", "Production", "2026-08-12"),
  mk("MP3-BG-28", "Bore gauge", "UNIT3", null, "2026-08-13"),
  mk("MP3-BG-29", "Bore gauge", "UNIT3", "QUALITY", "2026-08-13"),
  mk("MP3-PD-24", "Plunger dial", "UNIT3", "Production", "2026-08-13"),
  mk("MP3-CMM-01", "CMM", "UNIT3", "QUALITY", "2026-08-13"),
  mk("MP-QRQ-207 G", "QRG", "UNIT1", "Production", "2026-08-17"),
  mk("MP-PPG-13", "PPG", "UNIT1", "Production", "2026-08-17"),
  mk("MP-BMIC-02", "Blade Micrometer", "UNIT1", "Production", "2026-09-01"),
  mk("MP-DVC-67", "Digital vernier caliper", "UNIT1", "QUALITY", "2026-09-01"),
  mk("MP-DVC-68", "Digital vernier caliper", "UNIT1", null, "2026-09-01"),
];

const user: NotificationUser = { id: 1, name: "Admin", email: "a@b.c", roleName: "Tools Admin", unitScopes: ["COMMON"] };
const logo = "file://" + path.join(process.cwd(), "public", "email-logo.png");
const out = path.join(process.cwd(), "_to_delete", "emailpreview");
fs.writeFileSync(path.join(out, "digest.html"), digestTemplate(user, items, "D260819").replace(/cid:suki-tools-logo/g, logo));
fs.writeFileSync(path.join(out, "test.html"), testEmailTemplate(items[0]).replace(/cid:suki-tools-logo/g, logo));
