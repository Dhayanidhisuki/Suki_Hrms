 * Branded, responsive HTML email templates for SUKI ERP calibration
 * notifications.

 * All styles are inline (required for email clients).

 *

 * rendered at 3x from the same source artwork as `public/logo-icon-blue.svg`.
 */

/**

// ─── Helpers ───────────────────────────────────────────────────────────────────

export function escapeHtml(value: string): string {
  return value.replace(/[&<>'"]/g, (c) => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;", "'": "&#39;", '"': "&quot;",
  })[c]!);
}

function unitLabel(value: string | null) {
  return value?.replace(/^UNIT/, "Unit ") ?? "Not assigned";
}

function formatDate(date: Date): string {
  return date.toLocaleDateString("en-IN", {
    day: "2-digit", month: "short", year: "numeric",
  });
}

type StatusMeta = { label: string; color: string; bg: string; dot: string };

function statusMeta(item: DueCalibrationTool): StatusMeta {
  if (item.dueStatus === "OVERDUE") {
    const days = Math.abs(item.daysRemaining);
    return {
      label: `${days} day${days !== 1 ? "s" : ""} overdue`,
      color: BRAND.overdue,
      bg: "#FDF0F0",
      dot: "&#9679;",
    };
  }
  if (item.dueStatus === "DUE_TODAY") {
    return { label: "Due Today", color: BRAND.dueToday, bg: "#F7F0FE", dot: "&#9679;" };
  }
  return {
    label: `${item.daysRemaining} day${item.daysRemaining !== 1 ? "s" : ""} remaining`,
    color: BRAND.dueSoon,
    bg: "#FFF9E8",
    dot: "&#9679;",
  };
}

// ─── Digest template (one email per user, multiple tools) ──────────────────────

export function digestTemplate(
  user: NotificationUser,
  items: DueCalibrationTool[],
  channel: string,
): string {
  const overdue = items.filter((i) => i.dueStatus === "OVERDUE");
  const dueToday = items.filter((i) => i.dueStatus === "DUE_TODAY");
  const dueSoon = items.filter((i) => i.dueStatus === "DUE_SOON");

  const summaryCards = `
    <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="margin-bottom:20px;">
      <tr>
        ${summaryCard(overdue.length, "Overdue", BRAND.overdue, "#FDF0F0", "#F3B9B9")}
        ${summaryCard(dueToday.length, "Due Today", BRAND.dueToday, "#F7F0FE", "#D6BDF5")}
        ${summaryCard(dueSoon.length, "Due Soon", BRAND.dueSoon, "#FFF9E8", "#F0D48A")}
      </tr>
    </table>`;

  const th = (label: string, hideSm = false) =>
    `<th align="left" class="cell${hideSm ? " hide-sm" : ""}" style="padding:12px 10px;font-family:${BRAND.fontFamily};font-size:11px;font-weight:700;letter-spacing:.4px;text-transform:uppercase;color:${BRAND.navy};border-bottom:1px solid ${BRAND.borderLight};white-space:nowrap;">${escapeHtml(label)}</th>`;

  const tableRows = items.map((item, idx) => {
    const meta = statusMeta(item);
    const last = idx === items.length - 1;
    const border = last ? "" : `border-bottom:1px solid ${BRAND.borderRow};`;
    const td = (html: string, opts: { color?: string; nowrap?: boolean; hideSm?: boolean } = {}) =>
      `<td class="cell${opts.hideSm ? " hide-sm" : ""}" style="padding:12px 10px;${border}font-family:${BRAND.fontFamily};font-size:13px;font-weight:700;color:${opts.color ?? BRAND.textPrimary};${opts.nowrap ? "white-space:nowrap;" : ""}">${html}</td>`;
    return `
      <tr>
        ${td(escapeHtml(item.toolOrGaugeNo), { nowrap: true })}
        ${td(escapeHtml(item.name || item.description || "—"))}
        ${td(escapeHtml(unitLabel(item.unitCode)), { nowrap: true, hideSm: true })}
        ${td(escapeHtml(item.usedLocation || "—"), { hideSm: true })}
        ${td(formatDate(item.dueDate), { nowrap: true })}
        ${td(`${meta.dot}&nbsp;${escapeHtml(meta.label)}`, { color: meta.color, nowrap: true })}
      </tr>`;
  }).join("");

  // One download for the whole list — replaces the old per-row PDF links.
  const pdfUrl = calibrationDigestExportUrl(user.id, channel, "pdf");
  const xlsxUrl = calibrationDigestExportUrl(user.id, channel, "xlsx");
  const downloadBlock = pdfUrl && xlsxUrl
    ? `
    <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="border:1px solid ${BRAND.borderLight};border-radius:8px;margin-bottom:22px;">
      <tr>
        <td class="px" align="center" style="padding:18px 20px;">
          <p style="margin:0 0 12px;font-family:${BRAND.fontFamily};font-size:13px;color:${BRAND.textSecondary};">
            Download the full list of <strong style="color:${BRAND.navy};">${items.length} calibration record${items.length !== 1 ? "s" : ""}</strong>
          </p>
          <table role="presentation" cellpadding="0" cellspacing="0" border="0" style="display:inline-table;">
            <tr>
              <td style="padding-right:10px;">
                <a href="${escapeHtml(pdfUrl)}" style="display:inline-block;background-color:${BRAND.navyDeep};color:#FFFFFF;text-decoration:none;font-family:${BRAND.fontFamily};font-size:13px;font-weight:700;padding:11px 22px;border-radius:6px;white-space:nowrap;">&#128196;&nbsp; Download PDF</a>
              </td>
              <td>
                <a href="${escapeHtml(xlsxUrl)}" style="display:inline-block;background-color:#FFFFFF;border:1px solid ${BRAND.navy};color:${BRAND.navy};text-decoration:none;font-family:${BRAND.fontFamily};font-size:13px;font-weight:700;padding:10px 22px;border-radius:6px;white-space:nowrap;">&#128202;&nbsp; Download Excel</a>
              </td>
            </tr>
          </table>
        </td>
      </tr>
    </table>`
    : "";

  const content = `
    <!-- Greeting -->
    <p style="margin:0 0 10px;font-family:${BRAND.fontFamily};font-size:26px;font-weight:700;color:${BRAND.textPrimary};">Hello, ${escapeHtml(user.name)} &#128075;</p>
    <p style="margin:0 0 20px;font-family:${BRAND.fontFamily};font-size:15px;line-height:24px;color:${BRAND.textSecondary};">
      You have <strong style="color:${BRAND.navy};">${items.length} calibration reminder${items.length !== 1 ? "s" : ""}</strong> that match your role
      (<span style="color:${BRAND.navy};font-weight:700;">${escapeHtml(user.roleName)}</span>) and unit scope.<br />
      Please review and take action as needed.
    </p>

    ${summaryCards}

    <!-- Instruments table -->
    <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="border:1px solid ${BRAND.borderLight};border-radius:8px;border-collapse:separate;overflow:hidden;margin-bottom:22px;">
      <tr style="background-color:${BRAND.navyTint};">
        ${th("Tool / Gauge No.")}
        ${th("Tool Name")}
        ${th("Unit", true)}
        ${th("Used Location", true)}
        ${th("Due Date")}
        ${th("Status")}
      </tr>
      ${tableRows}
    </table>

    ${downloadBlock}

    <!-- Next step -->
    <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="background-color:${BRAND.navyTint};border-left:5px solid ${BRAND.navy};border-radius:6px;">
      <tr>
        <td valign="top" width="64" style="padding:18px 0 18px 16px;">
          <table role="presentation" cellpadding="0" cellspacing="0" border="0" style="background-color:#FFFFFF;border-radius:50%;">
            <tr><td align="center" valign="middle" width="44" height="44" style="width:44px;height:44px;font-size:20px;line-height:44px;color:${BRAND.navy};">&#128196;</td></tr>
          </table>
        </td>
        <td valign="middle" style="padding:18px 18px 18px 12px;font-family:${BRAND.fontFamily};font-size:14px;line-height:22px;color:${BRAND.textSecondary};">
          <strong style="color:${BRAND.navy};">Next step :</strong> Log into the ${escapeHtml(BRAND.name)} Tools Management System to schedule or record
          calibration for any overdue or approaching instruments. Overdue instruments must be prioritised immediately.
        </td>
      </tr>
    </table>`;

  const preview = overdue.length > 0
    ? `${overdue.length} instrument${overdue.length !== 1 ? "s" : ""} overdue — ${items.length} calibration reminder${items.length !== 1 ? "s" : ""} require your attention`
    : `${items.length} calibration reminder${items.length !== 1 ? "s" : ""} for your review`;

  return emailWrapper(content, preview);
}

// ─── Single-tool template ──────────────────────────────────────────────────────

export function singleToolTemplate(item: DueCalibrationTool): string {
  return emailWrapper(
    singleToolBody(item, "Calibration action is required for the instrument listed below."),
    `Calibration ${item.dueStatus === "OVERDUE" ? "overdue" : "due"}: ${item.toolOrGaugeNo}`,
  );
}

// ─── Test-mode template ────────────────────────────────────────────────────────

export function testEmailTemplate(item: DueCalibrationTool): string {
  const banner = `
    <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="margin-bottom:20px;">
      <tr>
        <td style="background-color:#FFF9E8;border:2px dashed ${BRAND.dueSoon};border-radius:8px;padding:14px 18px;text-align:center;">
          <p style="margin:0;font-family:${BRAND.fontFamily};font-size:14px;font-weight:700;color:#8A6508;">TEST EMAIL &mdash; This is a system configuration test.</p>
          <p style="margin:4px 0 0;font-family:${BRAND.fontFamily};font-size:12px;color:${BRAND.dueSoon};">No production notification was consumed. This tool&rsquo;s real reminder will send on schedule.</p>
        </td>
      </tr>
    </table>`;

  const closing = `
    <p style="margin:20px 0 0;font-family:${BRAND.fontFamily};font-size:13px;line-height:22px;color:${BRAND.textSecondary};">
      SMTP is configured correctly. If you received this email, delivery from
      <strong style="color:${BRAND.navy};">${escapeHtml(BRAND.name)} ${escapeHtml(BRAND.module)}</strong> is working as expected.
    </p>`;

  return emailWrapper(
    banner + singleToolBody(item, "Sample instrument used for SMTP delivery verification.") + closing,
    `[TEST] SMTP verification — ${item.toolOrGaugeNo} (${item.dueStatus.toLowerCase().replace("_", " ")})`,
  );
}

// ─── Private helpers ───────────────────────────────────────────────────────────

function singleToolBody(item: DueCalibrationTool, subtitle: string): string {
  const meta = statusMeta(item);
  const label = meta.label.charAt(0).toUpperCase() + meta.label.slice(1);
  return `
    <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="margin-bottom:22px;">
      <tr>
        <td style="background-color:${meta.bg};border-left:5px solid ${meta.color};border-radius:6px;padding:14px 18px;">
          <p style="margin:0;font-family:${BRAND.fontFamily};font-size:15px;font-weight:700;color:${meta.color};">${meta.dot}&nbsp; ${escapeHtml(label)}</p>
          <p style="margin:4px 0 0;font-family:${BRAND.fontFamily};font-size:13px;color:${BRAND.textSecondary};">${escapeHtml(subtitle)}</p>
        </td>
      </tr>
    </table>

    <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="border:1px solid ${BRAND.borderLight};border-radius:8px;border-collapse:separate;overflow:hidden;">
      <tr style="background-color:${BRAND.navyTint};">
        <td colspan="2" style="padding:12px 18px;border-bottom:1px solid ${BRAND.borderLight};font-family:${BRAND.fontFamily};font-size:11px;font-weight:700;letter-spacing:.6px;text-transform:uppercase;color:${BRAND.navy};">Instrument Details</td>
      </tr>
      ${detailRow("Tool / Gauge No.", escapeHtml(item.toolOrGaugeNo))}
      ${detailRow("Tool Name", escapeHtml(item.name || item.description || "—"))}
      ${detailRow("Size", escapeHtml(item.size || "—"))}
      ${detailRow("Unit", escapeHtml(unitLabel(item.unitCode)))}
      ${detailRow("Used Location", escapeHtml(item.usedLocation || "—"))}
      ${detailRow("Due Date", `<span style="color:${meta.color};font-weight:700;">${formatDate(item.dueDate)}</span>`)}
      ${detailRow("Status", `<span style="color:${meta.color};font-weight:700;">${meta.dot}&nbsp;${escapeHtml(meta.label)}</span>`, true)}
    </table>`;
}

function detailRow(label: string, valueHtml: string, last = false): string {
  const border = last ? "" : `border-bottom:1px solid ${BRAND.borderRow};`;
  return `
    <tr>
      <td style="padding:11px 18px;${border}font-family:${BRAND.fontFamily};font-size:12px;font-weight:700;color:${BRAND.navy};white-space:nowrap;width:170px;background-color:#FAFCFF;">${escapeHtml(label)}</td>
      <td style="padding:11px 18px;${border}font-family:${BRAND.fontFamily};font-size:13px;font-weight:700;color:${BRAND.textPrimary};">${valueHtml}</td>
    </tr>`;
}

function summaryCard(count: number, label: string, color: string, bg: string, border: string): string {
  return `
    <td class="stat" width="33.33%" valign="top" style="padding-right:8px;">
      <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="background-color:${bg};border:1px solid ${border};border-radius:10px;">
        <tr>
          <td align="center" style="padding:16px 8px;">
            <div style="font-family:${BRAND.fontFamily};font-size:32px;font-weight:700;line-height:36px;color:${color};">${count}</div>
            <div style="font-family:${BRAND.fontFamily};font-size:12px;font-weight:700;letter-spacing:.6px;text-transform:uppercase;color:${color};padding-top:4px;">${escapeHtml(label)}</div>
          </td>
        </tr>
      </table>
    </td>`;
}
